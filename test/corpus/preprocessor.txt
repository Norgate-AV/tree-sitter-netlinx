==================
Define Directives
==================

#DEFINE DEBUG_MODE
#define MAX_ATTEMPTS 5
#define DEVICE_NAME 'Touch Panel'

---

(source_file
  (preproc_directive
    (preproc_define
      (identifier)))
  (preproc_directive
    (preproc_define
      (identifier)
      (number_literal
        (decimal_literal))))
  (preproc_directive
    (preproc_define
      (identifier)
      (string_literal
        (string_content)))))

==================
Basic Conditional Directives
==================

#IF_NOT_DEFINED DEBUG_MODE
#DEFINE DEBUG_MODE
#END_IF

---

(source_file
  (preproc_directive
    (preproc_if_not_defined
      name: (identifier)
      (preproc_define
        (identifier))
      (preproc_end_if))))
